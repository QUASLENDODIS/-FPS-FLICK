local dragging, dragInput, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local function getCharacterFromInstance(instance)
    local current = instance
    while current and current ~= workspace do
        if current:FindFirstChildOfClass("Humanoid") then
            return current
        end
        current = current.Parent
    end
    return nil
end

local function fireAt(targetPart)
    local ClientRemotes = LocalPlayer:FindFirstChild("ClientRemotes")
    local ModuleScripts = ReplicatedStorage:FindFirstChild("ModuleScripts")
    if not ClientRemotes or not ModuleScripts then return end

    local Reload = ClientRemotes:FindFirstChild("Reload")
    local CheckFire = ClientRemotes:FindFirstChild("CheckFire")
    local CheckShot = ClientRemotes:FindFirstChild("CheckShot")
    local GunModules = ModuleScripts:FindFirstChild("GunModules")
    local ProjectileFinished = GunModules and GunModules.Remote:FindFirstChild("ProjectileFinished")
    local ProjectileRender = GunModules and GunModules.Remote:FindFirstChild("ProjectileRender")

    if not (CheckFire and CheckShot and ProjectileFinished and ProjectileRender) then return end

    local indexVal = tick() + math.random()
    local originPos = Camera.CFrame.Position
    local hitPos = targetPart.Position

    pcall(function()
        if Reload then
            Reload:FireServer()
        end

        CheckFire:FireServer(indexVal, originPos)

        ProjectileRender:FireServer({
            Type = "Bullet",
            Index = indexVal,
            Origin = originPos,
            Owner = LocalPlayer.Character,
            Endpoint = hitPos,
            Life = 5,
            Force = 130,
            Gravity = 0,
            Wind = Vector3.new(0, 0, 0)
        })

        local cf = Camera.CFrame
        CheckShot:FireServer(0, 0, 1, 0.8, cf, hitPos, targetPart, 11, indexVal)

        ProjectileFinished:FireServer({
            Index = indexVal,
            Origin = originPos,
            CFrame = CFrame.new(hitPos),
            ExpRadius = 15,
            LaserColor = Color3.new(0, 1, 1),
            LaserSpeed = 10,
            LaserTime = 0.7,
            Effect = "Gib_T",
            ExpColor = Color3.new(1, 0.6, 0.2),
            LaserTexture = "rbxassetid://4813348676",
            Explosive = false,
            HitPlayer = true,
            IsLaser = false,
            LaserWidth = 3,
            ExpSound = "rbxassetid://2814354338",
            LaserLength = 5,
            LaserEmission = 1
        })
    end)
end
-- Part 4: RunService RenderStepped Loop & Raycast Detection
local raycastParams = RaycastParams.new()
raycastParams.FilterType = Enum.RaycastFilterType.Exclude
raycastParams.IgnoreWater = true

RunService.RenderStepped:Connect(function()
    if isSpectating or (LocalPlayer.PlayerGui:FindFirstChild("Sp")) then
        StatusText.Text = "● Status: Spectating (Paused)"
        StatusText.TextColor3 = Color3.fromRGB(255, 90, 90)
        return
    else
        StatusText.Text = "● Status: Operational"
        StatusText.TextColor3 = Color3.fromRGB(0, 229, 140)
    end

    if not autoShootEnabled then return end
    if tick() - lastShotTime < fireCooldown then return end
    
    local character = LocalPlayer.Character
    if not character or not character:FindFirstChildOfClass("Humanoid") or character:FindFirstChildOfClass("Humanoid").Health <= 0 then 
        return 
    end

    raycastParams.FilterDescendantsInstances = {character}
    local viewportRay = Camera:ViewportPointToRay(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local result = workspace:Raycast(viewportRay.Origin, viewportRay.Direction * 10000, raycastParams)

    if result and result.Instance then
        local hitInstance = result.Instance
        local targetChar = getCharacterFromInstance(hitInstance)

        if targetChar then
            local humanoid = targetChar:FindFirstChildOfClass("Humanoid")
            local player = Players:GetPlayerFromCharacter(targetChar)

            if humanoid and humanoid.Health > 0 and player and player ~= LocalPlayer then
                lastShotTime = tick()
                local targetPart = targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart") or hitInstance
                fireAt(targetPart)
            end
        end
    end
end)
