        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

            Core.State.OriginalPosition = MainFrame.Position
        end
    end)

    return ScreenGui, MainFrame, GearIcon, TitleText, MinBtn, TitleBar

local Core = require(script.Parent.Part1_Core)
local S = Core.Services
local AnimController = {}

function AnimController.BindAnimations(MainFrame, GearIcon, TitleText, MinBtn, Container)
    local originalSize = Core.State.OriginalSize
    local closedSize = Core.State.ClosedSize

    local function toggleOpenClose()
        Core.State.IsOpen = not Core.State.IsOpen
        local currentPos = MainFrame.Position
        
        if not Core.State.IsOpen then
            -- 1. Instantly hide text and content
            Container.Visible = false
            TitleText.Visible = false
            MinBtn.Visible = false
            
            -- 2. Move UP slightly first
            local upPos = UDim2.new(currentPos.X.Scale, currentPos.X.Offset, currentPos.Y.Scale, currentPos.Y.Offset - 40)
            local tweenUp = S.TweenService:Create(MainFrame, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = upPos})
            tweenUp:Play()
            tweenUp.Completed:Wait()
            
            -- 3. Slide LEFT while spinning gear & shrinking to bevel box
            local leftPos = UDim2.new(currentPos.X.Scale, currentPos.X.Offset - 120, currentPos.Y.Scale, currentPos.Y.Offset - 40)
            local tweenSlide = S.TweenService:Create(MainFrame, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                Position = leftPos,
                Size = closedSize
            })
            local tweenGear = S.TweenService:Create(GearIcon, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                Rotation = GearIcon.Rotation - 360
            })
            tweenSlide:Play()
            tweenGear:Play()
        else
            -- Reverse: Expand back out to original position and size
            local tweenExpand = S.TweenService:Create(MainFrame, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                Position = Core.State.OriginalPosition,
                Size = originalSize
            })
            local tweenGear = S.TweenService:Create(GearIcon, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                Rotation = 0
            })
            tweenExpand:Play()
            tweenGear:Play()
            
            tweenExpand.Completed:Wait()
            Container.Visible = true
            TitleText.Visible = true
            MinBtn.Visible = true
            MinBtn.Text = "-"
        end
    end

    GearIcon.MouseButton1Click:Connect(toggleOpenClose)
    MinBtn.MouseButton1Click:Connect(toggleOpenClose)

    -- Startup Opening Animation
    task.spawn(function()
        MainFrame.Size = closedSize
        Container.Visible = false
        TitleText.Visible = false
        MinBtn.Visible = false
        GearIcon.Rotation = -360
        task.wait(0.2)
        
        Core.State.IsOpen = true
        Container.Visible = true
        TitleText.Visible = true
        MinBtn.Visible = true
        MinBtn.Text = "-"
        
        S.TweenService:Create(MainFrame, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = originalSize}):Play()
        S.TweenService:Create(GearIcon, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Rotation = 0}):Play()
    end)
end

return AnimController

local Core = require(script.Parent.Part1_Core)
local S = Core.Services
local Combat = {}

function Combat.Initialize(AutoTrack, AimTrack, SensBox, StatusText)
    -- Sensitivity Input Handling
    SensBox.FocusLost:Connect(function()
        local val = tonumber(SensBox.Text)
        if val then
            Core.Config.Sensitivity = math.max(0.01, val)
            SensBox.Text = tostring(Core.Config.Sensitivity)
        else
            SensBox.Text = tostring(Core.Config.Sensitivity)
        end
    end)

    -- Toggle Binding Helper
    local function bindToggle(trackButton, configKey)
        local state = false
        trackButton.MouseButton1Click:Connect(function()
            state = not state
            Core.Config[configKey] = state
            
            local knob = trackButton:FindFirstChildOfClass("Frame")
            local knobGoal = {}
            local trackGoal = {}
            local knobColorGoal = {}
            
            if state then
                knobGoal.Position = UDim2.new(1, -19, 0.5, -8)
                trackGoal.BackgroundColor3 = Color3.fromRGB(200, 40, 40)
                knobColorGoal.BackgroundColor3 = Core.Theme.TextPrimary
            else
                knobGoal.Position = UDim2.new(0, 3, 0.5, -8)
                trackGoal.BackgroundColor3 = Color3.fromRGB(35, 18, 18)
                knobColorGoal.BackgroundColor3 = Color3.fromRGB(160, 130, 130)
            end
            
            if knob then
                S.TweenService:Create(knob, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), knobGoal):Play()
                S.TweenService:Create(knob, TweenInfo.new(0.2), knobColorGoal):Play()
            end
            S.TweenService:Create(trackButton, TweenInfo.new(0.2), trackGoal):Play()
        end)
    end

    bindToggle(AutoTrack, "AutoShoot")
    bindToggle(AimTrack, "AimAssist")

    -- Spectator Check Hook
    local remotesFolder = S.ReplicatedStorage:FindFirstChild("Remotes")
    if remotesFolder then
        local startSpectatingRemote = remotesFolder:FindFirstChild("StartSpectating")
        if startSpectatingRemote then
            startSpectatingRemote.OnClientEvent:Connect(function()
                Core.State.IsSpectating = true
            end)
        end
    end

    Core.LocalPlayer.CharacterAdded:Connect(function()
        Core.State.IsSpectating = false
    end)

    local function getCharacterFromInstance(instance)
        local current = instance
        while current and current ~= workspace do
            if current:FindFirstChildOfClass("Humanoid") then
                return current
            end
            current = current.Parent
        end
        return nil
    end

    -- Wallcheck Helper
    local function isVisible(targetPart, targetChar)
        local origin = Core.Camera.CFrame.Position
        local direction = targetPart.Position - origin
        
        local raycastParams = RaycastParams.new()
        raycastParams.FilterType = Enum.RaycastFilterType.Exclude
        raycastParams.FilterDescendantsInstances = {Core.LocalPlayer.Character, targetChar}
        raycastParams.IgnoreWater = true
        
        local result = workspace:Raycast(origin, direction, raycastParams)
        if result then
            local hitChar = getCharacterFromInstance(result.Instance)
            if hitChar == targetChar then
                return true
            end
            return false
        end
        return true
    end

    -- Remote Firing Sequence
    local function fireAt(targetPart)
        local ClientRemotes = Core.LocalPlayer:FindFirstChild("ClientRemotes")
        local ModuleScripts = S.ReplicatedStorage:FindFirstChild("ModuleScripts")
        if not ClientRemotes or not ModuleScripts then return end

        local Reload = ClientRemotes:FindFirstChild("Reload")
        local CheckFire = ClientRemotes:FindFirstChild("CheckFire")
        local CheckShot = ClientRemotes:FindFirstChild("CheckShot")
        local GunModules = ModuleScripts:FindFirstChild("GunModules")
        local ProjectileFinished = GunModules and GunModules.Remote:FindFirstChild("ProjectileFinished")
        local ProjectileRender = GunModules and GunModules.Remote:FindFirstChild("ProjectileRender")

        if not (CheckFire and CheckShot and ProjectileFinished and ProjectileRender) then return end

        local indexVal = tick() + math.random()
        local originPos = Core.Camera.CFrame.Position
        local hitPos = targetPart.Position

        pcall(function()
            if Reload then Reload:FireServer() end
            CheckFire:FireServer(indexVal, originPos)

            ProjectileRender:FireServer({
                Type = "Bullet", Index = indexVal, Origin = originPos,
                Owner = Core.LocalPlayer.Character, Endpoint = hitPos,
                Life = 5, Force = 130, Gravity = 0, Wind = Vector3.new(0, 0, 0)
            })

            local cf = Core.Camera.CFrame
            CheckShot:FireServer(0, 0, 1, 0.8, cf, hitPos, targetPart, 11, indexVal)

            ProjectileFinished:FireServer({
                Index = indexVal, Origin = originPos, CFrame = CFrame.new(hitPos),
                ExpRadius = 15, LaserColor = Color3.new(1, 0, 0), LaserSpeed = 10,
                LaserTime = 0.7, Effect = "Gib_T", ExpColor = Core.Theme.AccentOrange,
                LaserTexture = "rbxassetid://4813348676", Explosive = false,
                HitPlayer = true, IsLaser = false, LaserWidth = 3,
                ExpSound = "rbxassetid://2814354338", LaserLength = 5, LaserEmission = 1
            })
        end)
    end

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.IgnoreWater = true

    -- Main Processing Loop
    S.RunService.RenderStepped:Connect(function()
        if Core.State.IsSpectating or (Core.LocalPlayer.PlayerGui:FindFirstChild("Sp")) then
            StatusText.Text = "● Status: Spectating (Paused)"
            StatusText.TextColor3 = Color3.fromRGB(255, 90, 90)
            return
        else
            StatusText.Text = "● Status: Operational"
            StatusText.TextColor3 = Color3.fromRGB(0, 229, 140)
        end

        local character = Core.LocalPlayer.Character
        if not character or not character:FindFirstChildOfClass("Humanoid") or character:FindFirstChildOfClass("Humanoid").Health <= 0 then 
            return 
        end

        -- Soft Aim Assist with Wallcheck
        if Core.Config.AimAssist then
            local closestTarget = nil
            local shortestDist = math.huge
            local mousePos = Vector2.new(Core.Camera.ViewportSize.X / 2, Core.Camera.ViewportSize.Y / 2)
            
            for _, player in ipairs(Core.Players:GetPlayers()) do
                if player ~= Core.LocalPlayer and player.Character then
                    local targetChar = player.Character
                    local hum = targetChar:FindFirstChildOfClass("Humanoid")
                    local head = targetChar:FindFirstChild("Head")
                    
                    if hum and hum.Health > 0 and head then
                        local screenPos, onScreen = Core.Camera:WorldToViewportPoint(head.Position)
                        if onScreen then
                            local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                            
                            if screenDist <= Core.Config.AimFOV and isVisible(head, targetChar) then
                                if screenDist < shortestDist then
                                    shortestDist = screenDist
                                    closestTarget = head
                                end
                            end
                        end
                    end
                end
            end
            
            if closestTarget then
                local targetCF = CFrame.new(Core.Camera.CFrame.Position, closestTarget.Position)
                local smoothFactor = math.clamp(0.02 * Core.Config.Sensitivity, 0.005, 1.0)
                Core.Camera.CFrame = Core.Camera.CFrame:Lerp(targetCF, smoothFactor)
            end
        end

        -- Auto-Shoot Triggerbot Loop
        if not Core.Config.AutoShoot then return end
        if tick() - Core.State.LastShotTime < Core.Config.FireCooldown then return end

        raycastParams.FilterDescendantsInstances = {character}
        local viewportRay = Core.Camera:ViewportPointToRay(Core.Camera.ViewportSize.X / 2, Core.Camera.ViewportSize.Y / 2)
        local result = workspace:Raycast(viewportRay.Origin, viewportRay.Direction * Core.Config.MaxDistance, raycastParams)

        if result and result.Instance then
            local hitInstance = result.Instance
            local targetChar = getCharacterFromInstance(hitInstance)

            if targetChar then
                local humanoid = targetChar:FindFirstChildOfClass("Humanoid")
                local player = Core.Players:GetPlayerFromCharacter(targetChar)

                if humanoid and humanoid.Health > 0 and player and player ~= Core.LocalPlayer then
                    Core.State.LastShotTime = tick()
                    local targetPart = targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart") or hitInstance
                    fireAt(targetPart)
                end
            end
        end
    end)
end

return Combat
