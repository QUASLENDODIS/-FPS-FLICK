local function toggleMinimize()
    State.IsMinimized = not State.IsMinimized
    
    local targetSize = State.IsMinimized and UDim2.new(0, 320, 0, 44) or UDim2.new(0, 320, 0, 310)
    local targetRotation = State.IsMinimized and GearIcon.Rotation + 360 or GearIcon.Rotation - 360
    
    if State.IsMinimized then
        Container.Visible = false
    else
        Container.Visible = true
    end
    
    TweenService:Create(MainFrame, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = targetSize}):Play()
    TweenService:Create(GearIcon, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Rotation = targetRotation}):Play()
    
    MinBtn.Text = State.IsMinimized and "+" or "-"
end

MinBtn.MouseButton1Click:Connect(toggleMinimize)

local function bindToggle(trackButton, configKey)
    local state = false
    trackButton.MouseButton1Click:Connect(function()
        state = not state
        Config[configKey] = state
        
        local knob = trackButton:FindFirstChildOfClass("Frame")
        local knobGoal = {}
        local trackGoal = {}
        local knobColorGoal = {}
        
        if state then
            knobGoal.Position = UDim2.new(1, -19, 0.5, -8)
            trackGoal.BackgroundColor3 = Color3.fromRGB(110, 40, 215)
            knobColorGoal.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        else
            knobGoal.Position = UDim2.new(0, 3, 0.5, -8)
            trackGoal.BackgroundColor3 = Color3.fromRGB(28, 28, 40)
            knobColorGoal.BackgroundColor3 = Color3.fromRGB(130, 130, 150)
        end
        
        if knob then
            TweenService:Create(knob, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), knobGoal):Play()
            TweenService:Create(knob, TweenInfo.new(0.2), knobColorGoal):Play()
        end
        TweenService:Create(trackButton, TweenInfo.new(0.2), trackGoal):Play()
    end)
end

local function getCharacterFromInstance(instance)
    local current = instance
    while current and current ~= workspace do
        if current:FindFirstChildOfClass("Humanoid") then
            return current
        end
        current = current.Parent
    end
    return nil
end

local function isVisible(targetPart, targetChar)
    local origin = Camera.CFrame.Position
    local direction = targetPart.Position - origin
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {LocalPlayer.Character, targetChar}
    raycastParams.IgnoreWater = true
    
    local result = workspace:Raycast(origin, direction, raycastParams)
    if result then
        local hitChar = getCharacterFromInstance(result.Instance)
        if hitChar == targetChar then
            return true
        end
        return false
    end
    return true
end

-- =================================================================
local function fireAt(targetPart)
    local ClientRemotes = LocalPlayer:FindFirstChild("ClientRemotes")
    local ModuleScripts = ReplicatedStorage:FindFirstChild("ModuleScripts")
    if not ClientRemotes or not ModuleScripts then return end

    local Reload = ClientRemotes:FindFirstChild("Reload")
    local CheckFire = ClientRemotes:FindFirstChild("CheckFire")
    local CheckShot = ClientRemotes:FindFirstChild("CheckShot")
    local GunModules = ModuleScripts:FindFirstChild("GunModules")
    local ProjectileFinished = GunModules and GunModules.Remote:FindFirstChild("ProjectileFinished")
    local ProjectileRender = GunModules and GunModules.Remote:FindFirstChild("ProjectileRender")

    if not (CheckFire and CheckShot and ProjectileFinished and ProjectileRender) then return end

    local indexVal = tick() + math.random()
    local originPos = Camera.CFrame.Position
    local hitPos = targetPart.Position

    pcall(function()
        if Reload then
            Reload:FireServer()
        end

        CheckFire:FireServer(indexVal, originPos)

        ProjectileRender:FireServer({
            Type = "Bullet",
            Index = indexVal,
            Origin = originPos,
            Owner = LocalPlayer.Character,
            Endpoint = hitPos,
            Life = 5,
            Force = 130,
            Gravity = 0,
            Wind = Vector3.new(0, 0, 0)
        })

        local cf = Camera.CFrame
        CheckShot:FireServer(0, 0, 1, 0.8, cf, hitPos, targetPart, 11, indexVal)

        ProjectileFinished:FireServer({
            Index = indexVal,
            Origin = originPos,
            CFrame = CFrame.new(hitPos),
            ExpRadius = 15,
            LaserColor = Color3.new(0, 1, 1),
            LaserSpeed = 10,
            LaserTime = 0.7,
            Effect = "Gib_T",
            ExpColor = Color3.new(1, 0.6, 0.2),
            LaserTexture = "rbxassetid://4813348676",
            Explosive = false,
            HitPlayer = true,
            IsLaser = false,
            LaserWidth = 3,
            ExpSound = "rbxassetid://2814354338",
            LaserLength = 5,
            LaserEmission = 1
        })
    end)
end

local raycastParams = RaycastParams.new()
raycastParams.FilterType = Enum.RaycastFilterType.Exclude
raycastParams.IgnoreWater = true

RunService.RenderStepped:Connect(function()
    if State.IsSpectating or (LocalPlayer.PlayerGui:FindFirstChild("Sp")) then
        StatusText.Text = "● Status: Spectating (Paused)"
        StatusText.TextColor3 = Color3.fromRGB(255, 90, 90)
        return
    else
        StatusText.Text = "● Status: Operational"
        StatusText.TextColor3 = Color3.fromRGB(0, 229, 140)
    end

    local character = LocalPlayer.Character
    if not character or not character:FindFirstChildOfClass("Humanoid") or character:FindFirstChildOfClass("Humanoid").Health <= 0 then 
        return 
    end

    -- Soft Aim Assist Processor
    if Config.AimAssist then
        local closestTarget = nil
        local shortestDist = math.huge
        local mousePos = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local targetChar = player.Character
                local hum = targetChar:FindFirstChildOfClass("Humanoid")
                local head = targetChar:FindFirstChild("Head")
                
                if hum and hum.Health > 0 and head then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(head.Position)
                    if onScreen then
                        local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                        
                        if screenDist <= Config.AimFOV and isVisible(head, targetChar) then
                            if screenDist < shortestDist then
                                shortestDist = screenDist
                                closestTarget = head
                            end
                        end
                    end
                end
            end
        end
        
        if closestTarget then
            local targetCF = CFrame.new(Camera.CFrame.Position, closestTarget.Position)
            local smoothFactor = math.clamp(0.02 * Config.Sensitivity, 0.005, 1.0)
            Camera.CFrame = Camera.CFrame:Lerp(targetCF, smoothFactor)
        end
    end

    -- Auto-Shoot Triggerbot Processor
    if not Config.AutoShoot then return end
    if tick() - State.LastShotTime < Config.FireCooldown then return end

    raycastParams.FilterDescendantsInstances = {character}
    local viewportRay = Camera:ViewportPointToRay(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local result = workspace:Raycast(viewportRay.Origin, viewportRay.Direction * Config.MaxDistance, raycastParams)

    if result and result.Instance then
        local hitInstance = result.Instance
        local targetChar = getCharacterFromInstance(hitInstance)

        if targetChar then
            local humanoid = targetChar:FindFirstChildOfClass("Humanoid")
            local player = Players:GetPlayerFromCharacter(targetChar)

            if humanoid and humanoid.Health > 0 and player and player ~= LocalPlayer then
                State.LastShotTime = tick()
                local targetPart = targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart") or hitInstance
                fireAt(targetPart)
            end
        end
    end
end)
