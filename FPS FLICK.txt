local Core = require(script.Parent.Core)

local Combat = {}

-- Helper to find player model from any hit part
local function getCharacterFromInstance(instance)
    local current = instance
    while current and current ~= workspace do
        if current:FindFirstChildOfClass("Humanoid") then
            return current
        end
        current = current.Parent
    end
    return nil
end

-- Full Remote Firing Sequence including Reload
function Combat.FireAt(targetPart)
    local ClientRemotes = Core.LocalPlayer:FindFirstChild("ClientRemotes")
    local ModuleScripts = S.ReplicatedStorage:FindFirstChild("ModuleScripts")
    if not ClientRemotes or not ModuleScripts then return end

    local Reload = ClientRemotes:FindFirstChild("Reload")
    local CheckFire = ClientRemotes:FindFirstChild("CheckFire")
    local CheckShot = ClientRemotes:FindFirstChild("CheckShot")
    local GunModules = ModuleScripts:FindFirstChild("GunModules")
    local ProjectileFinished = GunModules and GunModules.Remote:FindFirstChild("ProjectileFinished")
    local ProjectileRender = GunModules and GunModules.Remote:FindFirstChild("ProjectileRender")

    if not (CheckFire and CheckShot and ProjectileFinished and ProjectileRender) then return end

    local indexVal = tick() + math.random()
    local originPos = Core.Camera.CFrame.Position
    local hitPos = targetPart.Position

    pcall(function()
        if Reload then
            Reload:FireServer()
        end

        CheckFire:FireServer(indexVal, originPos)

        ProjectileRender:FireServer({
            Type = "Bullet",
            Index = indexVal,
            Origin = originPos,
            Owner = Core.LocalPlayer.Character,
            Endpoint = hitPos,
            Life = 5,
            Force = 130,
            Gravity = 0,
            Wind = Vector3.new(0, 0, 0)
        })

        local cf = Core.Camera.CFrame
        CheckShot:FireServer(0, 0, 1, 0.8, cf, hitPos, targetPart, 11, indexVal)

        ProjectileFinished:FireServer({
            Index = indexVal,
            Origin = originPos,
            CFrame = CFrame.new(hitPos),
            ExpRadius = 15,
            LaserColor = Color3.new(0, 1, 1),
            LaserSpeed = 10,
            LaserTime = 0.7,
            Effect = "Gib_T",
            ExpColor = Color3.new(1, 0.6, 0.2),
            LaserTexture = "rbxassetid://4813348676",
            Explosive = false,
            HitPlayer = true,
            IsLaser = false,
            LaserWidth = 3,
            ExpSound = "rbxassetid://2814354338",
            LaserLength = 5,
            LaserEmission = 1
        })
    end)
end

function Combat.InitializeLoops(statusLabelRef)
    -- Spectator Check Hook[span_0](start_span)[span_0](end_span)
    local remotesFolder = S.ReplicatedStorage:FindFirstChild("Remotes")
    if remotesFolder then
        local startSpectatingRemote = remotesFolder:FindFirstChild("StartSpectating")
        if startSpectatingRemote then
            startSpectatingRemote.OnClientEvent:Connect(function()
                Core.State.IsSpectating = true
            end)
        end
    end

    Core.LocalPlayer.CharacterAdded:Connect(function()
        Core.State.IsSpectating = false
    end)

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.IgnoreWater = true

    S.RunService.RenderStepped:Connect(function()
        if Core.State.IsSpectating or (Core.LocalPlayer.PlayerGui:FindFirstChild("Sp")) then
            if statusLabelRef then
                statusLabelRef.Text = "● Status: Spectating (Paused)"
                statusLabelRef.TextColor3 = Color3.fromRGB(255, 90, 90)
            end
            return
        else
            if statusLabelRef then
                statusLabelRef.Text = "● Status: Operational"
                statusLabelRef.TextColor3 = Color3.fromRGB(0, 229, 140)
            end
        end

        local character = Core.LocalPlayer.Character
        if not character or not character:FindFirstChildOfClass("Humanoid") or character:FindFirstChildOfClass("Humanoid").Health <= 0 then 
            return 
        end

        -- Aim Assist Processor
        if Core.Config.AimAssist then
            local closestTarget = nil
            local shortestDist = math.huge
            
            for _, player in ipairs(Core.Players:GetPlayers()) do
                if player ~= Core.LocalPlayer and player.Character then
                    local targetChar = player.Character
                    local hum = targetChar:FindFirstChildOfClass("Humanoid")
                    local head = targetChar:FindFirstChild("Head")
                    
                    if hum and hum.Health > 0 and head then
                        local screenPos, onScreen = Core.Camera:WorldToViewportPoint(head.Position)
                        if onScreen then
                            local mousePos = Vector2.new(Core.Camera.ViewportSize.X / 2, Core.Camera.ViewportSize.Y / 2)
                            local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                            
                            if dist < Core.Config.AimFOV and dist < shortestDist then
                                shortestDist = dist
                                closestTarget = head
                            end
                        end
                    end
                end
            end
            
            if closestTarget then
                local targetCF = CFrame.new(Core.Camera.CFrame.Position, closestTarget.Position)
                local smoothFactor = math.clamp(0.1 * Core.Config.Sensitivity, 0.01, 0.5)
                Core.Camera.CFrame = Core.Camera.CFrame:Lerp(targetCF, smoothFactor)
            end
        end

        -- Auto-Shoot Triggerbot Processor
        if not Core.Config.AutoShoot then return end
        if tick() - Core.State.LastShotTime < Core.Config.FireCooldown then return end

        raycastParams.FilterDescendantsInstances = {character}
        local viewportRay = Core.Camera:ViewportPointToRay(Core.Camera.ViewportSize.X / 2, Core.Camera.ViewportSize.Y / 2)
        local result = workspace:Raycast(viewportRay.Origin, viewportRay.Direction * Core.Config.MaxDistance, raycastParams)

        if result and result.Instance then
            local hitInstance = result.Instance
            local targetChar = getCharacterFromInstance(hitInstance)

            if targetChar then
                local humanoid = targetChar:FindFirstChildOfClass("Humanoid")
                local player = Core.Players:GetPlayerFromCharacter(targetChar)

                if humanoid and humanoid.Health > 0 and player and player ~= Core.LocalPlayer then
                    Core.State.LastShotTime = tick()
                    local targetPart = targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart") or hitInstance
                    Combat.FireAt(targetPart)
                end
            end
        end
    end)
end

return Combat

-- Toggle Animation Binding Helper
local function bindToggle(trackButton, configKey)
    local state = false
    trackButton.MouseButton1Click:Connect(function()
        state = not state
        Core.Config[configKey] = state
        
        local knob = trackButton:FindFirstChildOfClass("Frame")
        local knobGoal = {}
        local trackGoal = {}
        local knobColorGoal = {}
        
        if state then
            knobGoal.Position = UDim2.new(1, -19, 0.5, -8)
            trackGoal.BackgroundColor3 = Color3.fromRGB(110, 40, 215)
            knobColorGoal.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        else
            knobGoal.Position = UDim2.new(0, 3, 0.5, -8)
            trackGoal.BackgroundColor3 = Color3.fromRGB(28, 28, 40)
            knobColorGoal.BackgroundColor3 = Color3.fromRGB(130, 130, 150)
        end
        
        if knob then
            S.TweenService:Create(knob, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), knobGoal):Play()
            S.TweenService:Create(knob, TweenInfo.new(0.2), knobColorGoal):Play()
        end
        S.TweenService:Create(trackButton, TweenInfo.new(0.2), trackGoal):Play()
    end)
end
